Consultas que deciden la promocion

--- tasa-de-error   condicion de exito: result[0] <= 0.02
sum(
  rate(http_server_requests_total{
    namespace="{{args.namespace}}",
    servicio="{{args.servicio}}",
    version_pod="{{args.version-pod}}",
    familia="5xx"
  }[90s])
)
/
clamp_min(
  sum(
    rate(http_server_requests_total{
      namespace="{{args.namespace}}",
      servicio="{{args.servicio}}",
      version_pod="{{args.version-pod}}"
    }[90s])
  ),
  0.001
)
or on() vector(0)

--- latencia-p95   condicion de exito: result[0] <= 0.5
histogram_quantile(0.95,
  sum by (le) (
    rate(http_server_duration_seconds_bucket{
      namespace="{{args.namespace}}",
      servicio="{{args.servicio}}",
      version_pod="{{args.version-pod}}"
    }[90s])
  )
)
or on() vector(0)
